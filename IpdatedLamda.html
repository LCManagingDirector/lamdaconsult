// ── TABS

function showTab(id){

  document.querySelectorAll('.tab').forEach((t,i)=>{

    const ids=['housing','buildings','roads','bridges','water','liberia','pm'];

    t.classList.toggle('active',ids[i]===id);

  });

  document.querySelectorAll('.tab-panel').forEach(p=>p.classList.remove('active'));

  const el=document.getElementById('tab-'+id);

  if(el)el.classList.add('active');

}

// ── REVEAL ON SCROLL

const obs=new IntersectionObserver(entries=>{

  entries.forEach(e=>{if(e.isIntersecting)e.target.classList.add('on')});

},{threshold:0.1});

document.querySelectorAll('.reveal').forEach(el=>obs.observe(el));

// ── COUNTERS

function count(el){

  if(el.dataset.done)return;el.dataset.done=1;

  const t=parseInt(el.dataset.t),d=1600,s=performance.now();

  (function u(now){

    const p=Math.min((now-s)/d,1),e=1-Math.pow(1-p,3);

    el.textContent=Math.floor(e*t).toLocaleString();

    if(p<1)requestAnimationFrame(u);else el.textContent=t.toLocaleString();

  })(performance.now());

}

const cobs=new IntersectionObserver(e=>{

  e.forEach(x=>{if(x.isIntersecting)x.target.querySelectorAll('[data-t]').forEach(count)});

},{threshold:.4});

document.querySelectorAll('#stats').forEach(el=>cobs.observe(el));
